%-------------------------
% Reusable resume commands
%-------------------------

\newcommand{\resumeItem}[1]{
  \item\small{#1}
}

\newcommand{\resumeSubheading}[4]{
  \item
    \begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & {\small #4}\\[4pt]
      {\small #3} & {\small #2}\\
    \end{tabular*}\vspace{-2pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \item
    \begin{tabular*}{\textwidth}{l@{\extracolsep{\fill}}r}
      {\small #1} & {\small #2} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeProjectHeading}[2]{
    \item
    \begin{tabular*}{\textwidth}{l@{\extracolsep{\fill}}r}
      #1 & {\footnotesize #2} \\
    \end{tabular*}\vspace{4pt}
}

\newcommand{\resumeProjectMeta}[2]{
    {\small \hspace{0.22in}\textbf{#1}: #2\par}
}

\newcommand{\resumeSubItem}[1]{\resumeItem{#1}\vspace{-4pt}}

\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0in, label={}, itemsep=5pt, topsep=4pt]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}[leftmargin=0.34in, itemsep=4pt, topsep=4pt, parsep=0pt, partopsep=0pt]}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{2pt}}
\newcommand{\resumeProjectItemListStart}{\begin{itemize}[leftmargin=0.34in, itemsep=1pt, topsep=1pt, parsep=0pt, partopsep=0pt]}
\newcommand{\resumeProjectItemListEnd}{\end{itemize}\vspace{0pt}}
